pragma solidity 0.8.31;

import {Test, console2} from "forge-std/Test.sol";

interface IErc3009Token {
    function receiveWithAuthorization(address from, address to, uint256 value, uint256 validAfter, uint256 validBefore, bytes32 nonce, bytes memory signature) external;
    function receiveWithAuthorization(address from, address to, uint256 value, uint256 validAfter, uint256 validBefore, bytes32 nonce, uint8 v, bytes32 r, bytes32 s) external;
    function transferWithAuthorization(address from, address to, uint256 value, uint256 validAfter, uint256 validBefore, bytes32 nonce, bytes memory signature) external;
    function cancelAuthorization(address authorizer, bytes32 nonce, bytes memory signature) external;
    function authorizationState(address authorizer, bytes32 nonce) external view returns (bool);
    function DOMAIN_SEPARATOR() external view returns (bytes32);
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function transferFrom(address, address, uint256) external returns (bool);
    function approve(address, uint256) external returns (bool);
    function name() external view returns (string memory);
}
interface IERC1271 { function isValidSignature(bytes32 hash, bytes memory signature) external view returns (bytes4); }
interface ISimple7702Account {
    struct Call { address target; uint256 value; bytes data; }
    function execute(address target, uint256 value, bytes calldata data) external;
    function executeBatch(Call[] calldata calls) external;
    function entryPoint() external pure returns (address);
}

/// Minimal stand-in for the settlement router: path A pulls via ERC-3009, path B pulls via transferFrom(msg.sender).
contract MockRouter {
    uint256 transient locked; // TSTORE/TLOAD reentrancy guard (Cancun)
    IErc3009Token public immutable token;
    address public lastPayer; uint256 public lastAmount; uint256 public settlements;
    error Reentrancy();
    modifier nonReentrant() { if (locked == 1) revert Reentrancy(); locked = 1; _; locked = 0; }
    constructor(IErc3009Token t) { token = t; }
    function settleWithAuthorization(address from, uint256 value, uint256 validAfter, uint256 validBefore, bytes32 nonce, bytes calldata sig) external nonReentrant {
        token.receiveWithAuthorization(from, address(this), value, validAfter, validBefore, nonce, sig);
        lastPayer = from; lastAmount = value; settlements++;
    }
    function settleWithAuthorizationVRS(address from, uint256 value, uint256 validAfter, uint256 validBefore, bytes32 nonce, uint8 v, bytes32 r, bytes32 s) external nonReentrant {
        token.receiveWithAuthorization(from, address(this), value, validAfter, validBefore, nonce, v, r, s);
        lastPayer = from; lastAmount = value; settlements++;
    }
    function settle(uint256 value) external nonReentrant {
        token.transferFrom(msg.sender, address(this), value);
        lastPayer = msg.sender; lastAmount = value; settlements++;
    }
    function reenter() external nonReentrant { this.reenter(); }
}

contract PathsTest is Test {
    address constant AUSD = 0x00000000eFE302BEAA2b3e6e1b18d08D69a9012a;
    address constant USDC = 0x754704Bc059F8C67012fEd69BC8A327a5aafb603;
    address constant S7702 = 0xe6Cae83BdE06E4c305530e199D7217f42808555B;
    address constant EP08 = 0x4337084D9E255Ff0702461CF8895CE9E3b5Ff108;
    address constant POOL_AUSD = 0xb0a0264Ce6847F101b76ba36A4a3083ba489F501;
    address constant POOL_USDC = 0x463c0d1F04bcd99A1efCF94AC2a75bc19Ea4A7E5;
    bytes32 constant RECEIVE_TYPEHASH = 0xd099cc98ef71107a616c4f0f941f04c322d8e254fe26b3c6668db87aae413de8;
    bytes32 constant TRANSFER_TYPEHASH = 0x7c7c6cdb67a18743f49ec6fa9b35f50d52ed05cbed4cc592e13b44501c1a2267;
    bytes32 constant CANCEL_TYPEHASH = 0x158b0a9edf7a828aad02f63cd515c68ef2f50ba807396f6d12842833a1597429;
    bytes32 constant EIP712_DOMAIN_TYPEHASH = keccak256("EIP712Domain(string name,string version,uint256 chainId,address verifyingContract)");
    bytes32 constant INTENT_TYPEHASH = keccak256("PayoutIntent(address payer,address recipient,uint256 amountIn,bytes32 quoteId,uint256 deadline)");

    uint256 pk = 0xA11CE;
    address payer;
    MockRouter rA; MockRouter rU;

    function setUp() public {
        vm.createSelectFork("https://rpc.monad.xyz");
        payer = vm.addr(pk);
        rA = new MockRouter(IErc3009Token(AUSD));
        rU = new MockRouter(IErc3009Token(USDC));
        vm.prank(POOL_AUSD); IErc3009Token(AUSD).transfer(payer, 1_000e6);
        vm.prank(POOL_USDC); IErc3009Token(USDC).transfer(payer, 1_000e6);
        vm.deal(payer, 100 ether);
    }

    function digest(address token, bytes32 structHash) internal view returns (bytes32) {
        return keccak256(abi.encodePacked(hex"1901", IErc3009Token(token).DOMAIN_SEPARATOR(), structHash));
    }
    function receiveDigest(address token, address to, uint256 value, uint256 va, uint256 vb, bytes32 nonce) internal view returns (bytes32) {
        return digest(token, keccak256(abi.encode(RECEIVE_TYPEHASH, payer, to, value, va, vb, nonce)));
    }
    function sig(bytes32 d) internal view returns (uint8 v, bytes32 r, bytes32 s, bytes memory packed) {
        (v, r, s) = vm.sign(pk, d);
        packed = abi.encodePacked(r, s, v);
    }
    function delegate() internal {
        vm.signAndAttachDelegation(S7702, pk);
        (bool ok,) = payer.call("");
        require(ok, "call to payer failed");
        if (payer.code.length != 23) {
            console2.log("attachDelegation did not persist code; falling back to vm.etch");
            vm.etch(payer, abi.encodePacked(hex"ef0100", S7702));
        }
        assertEq(payer.code.length, 23, "delegated code length");
        assertEq(bytes3(payer.code), bytes3(0xef0100), "delegation prefix");
        assertEq(address(bytes20(bytes23(payer.code) << 24)), S7702, "delegate address");
        console2.log("payer.code.length after delegation:", payer.code.length);
    }

    function test_domainSeparators() public view {
        assertEq(block.chainid, 143);
        bytes32 a = keccak256(abi.encode(EIP712_DOMAIN_TYPEHASH, keccak256("Agora Dollar"), keccak256("1"), uint256(143), AUSD));
        assertEq(IErc3009Token(AUSD).DOMAIN_SEPARATOR(), a, "AUSD domain = (Agora Dollar, 1, 143, proxy)");
        bytes32 u = keccak256(abi.encode(EIP712_DOMAIN_TYPEHASH, keccak256("USDC"), keccak256("2"), uint256(143), USDC));
        assertEq(IErc3009Token(USDC).DOMAIN_SEPARATOR(), u, "USDC domain = (USDC, 2, 143, proxy)");
    }

    function _pathA(address token, MockRouter r, bytes32 n1, bytes32 n2) internal {
        uint256 va = 0; uint256 vb = block.timestamp + 1 hours;
        uint256 before = IErc3009Token(token).balanceOf(address(r));
        // bytes overload
        (,,, bytes memory packed) = sig(receiveDigest(token, address(r), 10e6, va, vb, n1));
        r.settleWithAuthorization(payer, 10e6, va, vb, n1, packed);
        assertEq(IErc3009Token(token).balanceOf(address(r)), before + 10e6);
        assertTrue(IErc3009Token(token).authorizationState(payer, n1));
        // replay -> revert
        vm.expectRevert();
        r.settleWithAuthorization(payer, 10e6, va, vb, n1, packed);
        // v,r,s overload
        (uint8 v, bytes32 rr, bytes32 ss,) = sig(receiveDigest(token, address(r), 7e6, va, vb, n2));
        r.settleWithAuthorizationVRS(payer, 7e6, va, vb, n2, v, rr, ss);
        assertEq(IErc3009Token(token).balanceOf(address(r)), before + 17e6);
        assertEq(r.lastPayer(), payer);
    }

    function test_pathA_plainEOA_AUSD() public { assertEq(payer.code.length, 0); _pathA(AUSD, rA, keccak256("a1"), keccak256("a2")); }
    function test_pathA_plainEOA_USDC() public { assertEq(payer.code.length, 0); _pathA(USDC, rU, keccak256("u1"), keccak256("u2")); }

    function test_pathA_delegatedEOA_AUSD() public {
        delegate();
        bytes32 d = receiveDigest(AUSD, address(rA), 10e6, 0, block.timestamp + 1 hours, keccak256("d1"));
        (,,, bytes memory packed) = sig(d);
        assertEq(IERC1271(payer).isValidSignature(d, packed), bytes4(0x1626ba7e), "Simple7702Account ERC-1271 accepts own ECDSA sig");
        _pathA(AUSD, rA, keccak256("d1"), keccak256("d2"));
    }
    function test_pathA_delegatedEOA_USDC() public {
        delegate();
        _pathA(USDC, rU, keccak256("du1"), keccak256("du2"));
    }
    function test_pathA_delegatedEOA_badSigRejected() public {
        delegate();
        uint256 vb = block.timestamp + 1 hours;
        bytes32 d = receiveDigest(AUSD, address(rA), 10e6, 0, vb, keccak256("bad"));
        (uint8 v, bytes32 r, bytes32 s) = vm.sign(0xB0B, d); // wrong key
        vm.expectRevert();
        rA.settleWithAuthorization(payer, 10e6, 0, vb, keccak256("bad"), abi.encodePacked(r, s, v));
        vm.expectRevert();
        rU.settleWithAuthorization(payer, 10e6, 0, vb, keccak256("bad"), abi.encodePacked(r, s, v));
    }

    function test_pathB_delegated_executeBatch() public {
        delegate();
        ISimple7702Account.Call[] memory calls = new ISimple7702Account.Call[](2);
        calls[0] = ISimple7702Account.Call(AUSD, 0, abi.encodeCall(IErc3009Token.approve, (address(rA), 25e6)));
        calls[1] = ISimple7702Account.Call(address(rA), 0, abi.encodeCall(MockRouter.settle, (25e6)));
        // from EntryPoint v0.8 (as during a sponsored userOp)
        vm.prank(EP08);
        ISimple7702Account(payer).executeBatch(calls);
        assertEq(rA.lastPayer(), payer, "router sees msg.sender == payer EOA");
        assertEq(IErc3009Token(AUSD).balanceOf(address(rA)), 25e6);
        // from self (EOA sending its own tx)
        vm.prank(payer);
        ISimple7702Account(payer).executeBatch(calls);
        assertEq(rA.settlements(), 2);
        // from a stranger -> rejected
        vm.prank(address(0xBEEF));
        vm.expectRevert(bytes("not from self or EntryPoint"));
        ISimple7702Account(payer).executeBatch(calls);
        assertEq(ISimple7702Account(payer).entryPoint(), EP08);
    }

    function test_nonceBindsIntent_and_cancel() public {
        uint256 vb = block.timestamp + 1 hours;
        bytes32 intent = keccak256(abi.encode(INTENT_TYPEHASH, payer, address(0xCAFE), uint256(10e6), keccak256("quote"), vb));
        (,,, bytes memory packed) = sig(receiveDigest(AUSD, address(rA), 10e6, 0, vb, intent));
        // tampered intent -> different nonce -> InvalidSignature
        bytes32 tampered = keccak256(abi.encode(INTENT_TYPEHASH, payer, address(0xCAFE), uint256(11e6), keccak256("quote"), vb));
        // token frame reverts InvalidSignature() (see -vvvv trace); forge fork backend rewrites the bubbled reason
        // when Solady ERC-1271-falls back to a code-less EOA, so match any revert here.
        vm.expectRevert();
        rA.settleWithAuthorization(payer, 10e6, 0, vb, tampered, packed);
        // correct intent settles
        rA.settleWithAuthorization(payer, 10e6, 0, vb, intent, packed);
        assertTrue(IErc3009Token(AUSD).authorizationState(payer, intent));
        // cancel another nonce before use (anyone can submit the payer-signed cancel)
        bytes32 n2 = keccak256("to-cancel");
        (,,, bytes memory csig) = sig(digest(AUSD, keccak256(abi.encode(CANCEL_TYPEHASH, payer, n2))));
        vm.prank(address(0xD00D));
        IErc3009Token(AUSD).cancelAuthorization(payer, n2, csig);
        assertTrue(IErc3009Token(AUSD).authorizationState(payer, n2));
        (,,, bytes memory p2) = sig(receiveDigest(AUSD, address(rA), 1e6, 0, vb, n2));
        vm.expectRevert(abi.encodeWithSignature("UsedOrCanceledAuthorization()"));
        rA.settleWithAuthorization(payer, 1e6, 0, vb, n2, p2);
    }

    function test_sameNonce_independentPerToken() public {
        uint256 vb = block.timestamp + 1 hours; bytes32 n = keccak256("shared");
        (,,, bytes memory pa) = sig(receiveDigest(AUSD, address(rA), 1e6, 0, vb, n));
        (,,, bytes memory pu) = sig(receiveDigest(USDC, address(rU), 1e6, 0, vb, n));
        rA.settleWithAuthorization(payer, 1e6, 0, vb, n, pa);
        rU.settleWithAuthorization(payer, 1e6, 0, vb, n, pu);
        // AUSD signature is NOT valid on USDC (different domain separator) even with same params
        vm.expectRevert();
        rU.settleWithAuthorization(payer, 1e6, 0, vb, keccak256("other"), pa);
    }

    function test_wrongPayeeRejected() public {
        uint256 vb = block.timestamp + 1 hours; bytes32 n = keccak256("payee");
        (,,, bytes memory p) = sig(receiveDigest(AUSD, address(rA), 1e6, 0, vb, n));
        vm.expectRevert(abi.encodeWithSignature("InvalidPayee(address,address)", address(this), address(rA)));
        IErc3009Token(AUSD).receiveWithAuthorization(payer, address(rA), 1e6, 0, vb, n, p);
        // transferWithAuthorization has no payee==msg.sender check: anyone can submit it
        bytes32 n3 = keccak256("twa");
        bytes32 d3 = digest(AUSD, keccak256(abi.encode(TRANSFER_TYPEHASH, payer, address(rA), uint256(1e6), uint256(0), vb, n3)));
        (,,, bytes memory p3) = sig(d3);
        vm.prank(address(0xD00D));
        IErc3009Token(AUSD).transferWithAuthorization(payer, address(rA), 1e6, 0, vb, n3, p3);
        assertEq(IErc3009Token(AUSD).balanceOf(address(rA)), 1e6);
    }

    function test_transientGuard() public {
        vm.expectRevert(MockRouter.Reentrancy.selector);
        rA.reenter();
    }
}
