// ---- ERC-4337 v0.8 (verbatim from Sourcify-verified EntryPoint 0x4337084D9E255Ff0702461CF8895CE9E3b5Ff108, chain 143) ----
struct PackedUserOperation {
    address sender;
    uint256 nonce;
    bytes initCode;
    bytes callData;
    bytes32 accountGasLimits;
    uint256 preVerificationGas;
    bytes32 gasFees;
    bytes paymasterAndData;
    bytes signature;
}
interface IEntryPoint {
    event UserOperationEvent(bytes32 indexed userOpHash, address indexed sender, address indexed paymaster, uint256 nonce, bool success, uint256 actualGasCost, uint256 actualGasUsed);
    event UserOperationRevertReason(bytes32 indexed userOpHash, address indexed sender, uint256 nonce, bytes revertReason);
    event BeforeExecution();
    error FailedOp(uint256 opIndex, string reason);
    error FailedOpWithRevert(uint256 opIndex, string reason, bytes inner);
    function handleOps(PackedUserOperation[] calldata ops, address payable beneficiary) external;
    function getUserOpHash(PackedUserOperation calldata userOp) external view returns (bytes32);
    function getNonce(address sender, uint192 key) external view returns (uint256 nonce);
    function depositTo(address account) external payable;
    function balanceOf(address account) external view returns (uint256);
    function getDomainSeparatorV4() external view returns (bytes32);
    function getPackedUserOpTypeHash() external pure returns (bytes32);
}
interface IPaymaster {
    enum PostOpMode { opSucceeded, opReverted, postOpReverted }
    function validatePaymasterUserOp(PackedUserOperation calldata userOp, bytes32 userOpHash, uint256 maxCost) external returns (bytes memory context, uint256 validationData);
    function postOp(PostOpMode mode, bytes calldata context, uint256 actualGasCost, uint256 actualUserOpFeePerGas) external;
}
interface ISimple7702Account {
    struct Call { address target; uint256 value; bytes data; }
    function execute(address target, uint256 value, bytes calldata data) external;
    function executeBatch(Call[] calldata calls) external;
}
interface IERC20 {
    function balanceOf(address) external view returns (uint256);
    function transfer(address, uint256) external returns (bool);
    function transferFrom(address, address, uint256) external returns (bool);
    function approve(address, uint256) external returns (bool);
    function allowance(address, address) external view returns (uint256);
}
// Mento V3 Router (verified on Monad Sourcify; see prior findings)
interface IMentoRouter {
    struct Route { address from; address to; address factory; }
    function getAmountsOut(uint256 amountIn, Route[] calldata routes) external view returns (uint256[] memory amounts);
    function swapExactTokensForTokens(uint256 amountIn, uint256 amountOutMin, Route[] calldata routes, address to, uint256 deadline) external returns (uint256[] memory amounts);
}

/// Stand-in for the Henad settlement router: pulls from msg.sender and records who that was.
contract MockRouter {
    IERC20 public immutable token;
    address public lastPayer; uint256 public lastAmount; uint256 public settlements;
    constructor(IERC20 t) { token = t; }
    function settle(uint256 value) external {
        require(token.transferFrom(msg.sender, address(this), value), "pull failed");
        lastPayer = msg.sender; lastAmount = value; settlements++;
    }
}

contract BaseFeeProbe { function basefee() external view returns (uint256) { return block.basefee; } function gasprice() external view returns (uint256) { return tx.gasprice; } }
contract Caller { function callTarget(address t, bytes calldata d) external returns (bytes memory r) { bool ok; (ok, r) = t.call(d); require(ok, "callTarget failed"); } }
contract DumbAccount {
    function validateUserOp(PackedUserOperation calldata, bytes32, uint256) external pure returns (uint256) { return 0; }
    fallback() external payable {}
    receive() external payable {}
}
contract DepthProbe {
    BaseFeeProbe immutable p;
    constructor(BaseFeeProbe _p) { p = _p; }
    function inner() external view returns (uint256) { return block.basefee; }
    function viaCall() external view returns (uint256) { return p.basefee(); }
    function viaSelf() external view returns (uint256) { return this.inner(); }
    function viaSelfThenCall() external view returns (uint256) { return this.viaCall(); }
}

/// Minimal sponsoring paymaster: accepts every op, records postOp.
contract StubPaymaster is IPaymaster {
    address public immutable ep;
    uint256 public validations; uint256 public postOps; PostOpMode public lastMode; uint256 public lastActualGasCost; uint256 public lastFeePerGas; uint256 public lastMaxCost;
    bool public reject;
    constructor(address _ep) { ep = _ep; }
    function setReject(bool r) external { reject = r; }
    function validatePaymasterUserOp(PackedUserOperation calldata, bytes32, uint256 maxCost) external override returns (bytes memory context, uint256 validationData) {
        require(msg.sender == ep, "pm: not EP");
        validations++; lastMaxCost = maxCost;
        // non-empty context so EntryPoint calls postOp (empty context = no postOp)
        return (abi.encode(uint256(42)), reject ? 1 : 0);
    }
    function postOp(PostOpMode mode, bytes calldata, uint256 actualGasCost, uint256 actualUserOpFeePerGas) external override {
        require(msg.sender == ep, "pm: not EP");
        postOps++; lastMode = mode; lastActualGasCost = actualGasCost; lastFeePerGas = actualUserOpFeePerGas;
    }
}

    // ---------- helpers ----------
    function delegatePayer() internal {
        // sign an EIP-7702 authorization for payer -> Simple7702Account and apply it via a type-4 tx
        Vm.SignedDelegation memory d = vm.signDelegation(S7702, pk);
        assertEq(d.implementation, S7702);

        assertEq(bytes3(payer.code), bytes3(0xef0100));
        assertEq(address(bytes20(bytes23(payer.code) << 24)), S7702);
    }
    function pack(uint256 hi, uint256 lo) internal pure returns (bytes32) { return bytes32((hi << 128) | lo); }
    function batchCalldata(ISimple7702Account.Call[] memory calls) internal pure returns (bytes memory) {
        return abi.encodeCall(ISimple7702Account.executeBatch, (calls));
    }
    function mockCalls(uint256 amount) internal view returns (ISimple7702Account.Call[] memory calls) {
        calls = new ISimple7702Account.Call[](2);
        calls[0] = ISimple7702Account.Call(AUSD, 0, abi.encodeCall(IERC20.approve, (address(router), amount)));
        calls[1] = ISimple7702Account.Call(address(router), 0, abi.encodeCall(MockRouter.settle, (amount)));
    }
    function buildOp(bytes memory initCode, bytes memory callData) internal view returns (PackedUserOperation memory op) {
        op = PackedUserOperation({
            sender: payer,
            nonce: EP.getNonce(payer, 0),
            initCode: initCode,
            callData: callData,
            accountGasLimits: pack(VERIF_GAS, CALL_GAS),
            preVerificationGas: PRE_VERIF_GAS,
            gasFees: pack(MAX_PRIO, MAX_FEE),
            paymasterAndData: abi.encodePacked(address(pm), uint128(PM_VERIF_GAS), uint128(PM_POSTOP_GAS)),
            signature: ""
        });
    }
    function signOp(PackedUserOperation memory op, bytes32 hash, uint256 key) internal pure returns (PackedUserOperation memory) {
        (uint8 v, bytes32 r, bytes32 s) = vm.sign(key, hash);
        op.signature = abi.encodePacked(r, s, v);
        return op;
    }
    /// Local re-implementation of EntryPoint v0.8 getUserOpHash (with the EIP-7702 initCode override),
    /// needed for the 0x7702 marker case where the hash must be signed BEFORE the sender has code.
    function localUserOpHash(PackedUserOperation memory op, address delegateForOverride) internal view returns (bytes32) {
        bytes32 initHash;
        if (op.initCode.length >= 2 && bytes2(op.initCode) == 0x7702 && op.initCode.length <= 20) {
            initHash = keccak256(abi.encodePacked(delegateForOverride));
        } else {
            initHash = keccak256(op.initCode);
        }
        bytes32 structHash = keccak256(abi.encode(
            EP.getPackedUserOpTypeHash(), op.sender, op.nonce, initHash, keccak256(op.callData),
            op.accountGasLimits, op.preVerificationGas, op.gasFees, keccak256(op.paymasterAndData)));
        return keccak256(abi.encodePacked(hex"1901", EP.getDomainSeparatorV4(), structHash));
    }
    function submit(PackedUserOperation memory op) internal returns (OpResult memory res) {
        PackedUserOperation[] memory ops = new PackedUserOperation[](1);
        ops[0] = op;
        vm.recordLogs();
        vm.prank(bundler, bundler);
        EP.handleOps(ops, payable(bundler));
        Vm.Log[] memory logs = vm.getRecordedLogs();
        for (uint256 i = 0; i < logs.length; i++) {
            if (logs[i].emitter != address(EP)) continue;
            if (logs[i].topics[0] == USEROP_EVENT_SIG) {
                res.found = true;
                res.paymaster = address(uint160(uint256(logs[i].topics[3])));
                (res.nonce, res.success, res.actualGasCost, res.actualGasUsed) = abi.decode(logs[i].data, (uint256, bool, uint256, uint256));
                assertEq(address(uint160(uint256(logs[i].topics[2]))), payer, "event sender");
            } else if (logs[i].topics[0] == REVERT_REASON_SIG) {
                res.revertReasonEmitted = true;
                (, res.revertReason) = abi.decode(logs[i].data, (uint256, bytes));
            }
        }
    }

    // ---------- sanity: the EntryPoint on the fork is v0.8 with the ERC4337/1 domain ----------
    function test_entryPointDomain() public view {
        bytes32 dom = keccak256(abi.encode(keccak256("EIP712Domain(string name,string version,uint256 chainId,address verifyingContract)"),
            keccak256("ERC4337"), keccak256("1"), uint256(143), address(EP)));
        assertEq(EP.getDomainSeparatorV4(), dom, "EntryPoint v0.8 EIP-712 domain (ERC4337, 1, 143, EP)");
        assertEq(EP.getPackedUserOpTypeHash(), keccak256("PackedUserOperation(address sender,uint256 nonce,bytes initCode,bytes callData,bytes32 accountGasLimits,uint256 preVerificationGas,bytes32 gasFees,bytes paymasterAndData)"));
        assertEq(address(EP).code.length > 0, true);
        assertEq(S7702.code.length > 0, true);
    }

    // ---------- 1. pre-delegated payer with ZERO MON, empty initCode, paymaster sponsors: executeBatch([approve, settle]) ----------
    function test_handleOps_preDelegated_zeroMon_mockRouter() public {
        delegatePayer();
        vm.deal(payer, 0);
        uint256 amount = 25e6;
        PackedUserOperation memory op = buildOp("", batchCalldata(mockCalls(amount)));
        bytes32 h = EP.getUserOpHash(op);
        assertEq(h, localUserOpHash(op, address(0)), "local hash == EP.getUserOpHash (empty initCode)");
        op = signOp(op, h, pk);

        uint256 nonceBefore = EP.getNonce(payer, 0);
        uint256 pmDepBefore = EP.balanceOf(address(pm));
        uint256 bundlerBefore = bundler.balance;
        uint256 payerAusdBefore = IERC20(AUSD).balanceOf(payer);

        OpResult memory r = submit(op);
        assertTrue(r.found, "UserOperationEvent emitted");
        assertTrue(r.success, "userOp success == true");
        assertFalse(r.revertReasonEmitted, "no UserOperationRevertReason");
        assertEq(r.nonce, nonceBefore);
        assertEq(r.paymaster, address(pm));
        assertEq(EP.getNonce(payer, 0), nonceBefore + 1, "EP nonce consumed");
        // router saw msg.sender == payer EOA, tokens moved
        assertEq(router.lastPayer(), payer, "router msg.sender == payer");
        assertEq(router.lastAmount(), amount);
        assertEq(IERC20(AUSD).balanceOf(address(router)), amount);
        assertEq(IERC20(AUSD).balanceOf(payer), payerAusdBefore - amount);
        assertEq(IERC20(AUSD).allowance(payer, address(router)), 0, "allowance fully consumed");
        // payer paid nothing in MON; paymaster deposit paid; bundler received exactly that
        assertEq(payer.balance, 0, "payer MON unchanged (still 0)");
        assertEq(pmDepBefore - EP.balanceOf(address(pm)), r.actualGasCost, "paymaster deposit charged actualGasCost");
        assertEq(bundler.balance - bundlerBefore, r.actualGasCost, "beneficiary received actualGasCost");
        assertEq(pm.validations(), 1); assertEq(pm.postOps(), 1);
        assertEq(uint8(pm.lastMode()), uint8(IPaymaster.PostOpMode.opSucceeded));
        console2.log("actualGasUsed", r.actualGasUsed, "actualGasCost(wei)", r.actualGasCost);
        console2.log("pm.lastFeePerGas", pm.lastFeePerGas(), "block.basefee in test frame", block.basefee);
        BaseFeeProbe probe = new BaseFeeProbe();
        console2.log("basefee seen by a called contract", probe.basefee(), "tx.gasprice", probe.gasprice());
        assertEq(r.actualGasCost, r.actualGasUsed * pm.lastFeePerGas(), "actualGasCost == actualGasUsed * userOpGasPrice");
        // maxCost passed to the paymaster = (verif+call+pmVerif+pmPostOp+preVerif) * maxFeePerGas -> proves which half holds maxFeePerGas
        uint256 totalGas = VERIF_GAS + CALL_GAS + PM_VERIF_GAS + PM_POSTOP_GAS + PRE_VERIF_GAS;
        console2.log("pm.lastMaxCost", pm.lastMaxCost(), "expected totalGas*MAX_FEE", totalGas * MAX_FEE);
        assertEq(pm.lastMaxCost(), totalGas * MAX_FEE, "maxFeePerGas is the LOW 128 bits of gasFees");
        // NOTE (fork artefact, see test_diag_epFrameBasefee_prioZero): inside the fork-loaded EntryPoint, block.basefee reads 0,
        // so the userOp gas price collapses to maxPriorityFeePerGas. On mainnet it is min(maxFee, prio + 100 gwei basefee).
        assertTrue(pm.lastFeePerGas() == MAX_PRIO || pm.lastFeePerGas() == MAX_PRIO + probe.basefee(), "userOp gas price = min(maxFee, prio+basefee)");
    }

    // ---------- 2. same, but the batch hits the REAL Mento Router: AUSD -> USDm -> GBPm ----------
    function test_handleOps_preDelegated_realMentoSwap() public {
        delegatePayer();
        vm.deal(payer, 5 ether); // below Monad's 10-MON reserve; must be untouched
        uint256 amountIn = 10e6;
        IMentoRouter.Route[] memory routes = new IMentoRouter.Route[](2);
        routes[0] = IMentoRouter.Route(AUSD, USDM, address(0));
        routes[1] = IMentoRouter.Route(USDM, GBPM, address(0));
        uint256[] memory q = IMentoRouter(MENTO_ROUTER).getAmountsOut(amountIn, routes);
        console2.log("quote: 10 AUSD ->", q[1], "USDm ->", q[2]);
        ISimple7702Account.Call[] memory calls = new ISimple7702Account.Call[](2);
        calls[0] = ISimple7702Account.Call(AUSD, 0, abi.encodeCall(IERC20.approve, (MENTO_ROUTER, amountIn)));
        calls[1] = ISimple7702Account.Call(MENTO_ROUTER, 0, abi.encodeCall(IMentoRouter.swapExactTokensForTokens, (amountIn, q[2], routes, recipient, block.timestamp + 1 hours)));
        PackedUserOperation memory op = buildOp("", batchCalldata(calls));
        op = signOp(op, EP.getUserOpHash(op), pk);

        uint256 payerMon = payer.balance;
        uint256 payerAusd = IERC20(AUSD).balanceOf(payer);
        OpResult memory r = submit(op);
        console2.log("found", r.found, "success", r.success);
        console2.log("actualGasUsed", r.actualGasUsed);
        if (r.revertReasonEmitted) {
            console2.logBytes(r.revertReason);
            if (bytes4(r.revertReason) == bytes4(keccak256("ExecuteError(uint256,bytes)"))) {
                bytes memory tail = new bytes(r.revertReason.length - 4);
                for (uint256 i = 0; i < tail.length; i++) tail[i] = r.revertReason[i + 4];
                (uint256 idx, bytes memory inner) = abi.decode(tail, (uint256, bytes));
                console2.log("ExecuteError index", idx); console2.logBytes(inner);
            }
        }
        assertTrue(r.found && r.success, "sponsored swap userOp succeeded");
        assertEq(IERC20(GBPM).balanceOf(recipient), q[2], "recipient got exactly the same-block quote");
        assertEq(IERC20(AUSD).balanceOf(payer), payerAusd - amountIn, "AUSD pulled from payer by Router.transferFrom(msg.sender)");
        assertEq(payer.balance, payerMon, "payer MON unchanged");
        assertEq(EP.getNonce(payer, 0), 1);
        console2.log("actualGasUsed", r.actualGasUsed, "GBPm out", q[2]);
    }

    function _c3(address t, bytes memory data) internal pure returns (bytes memory) {
        // Call3 { address target; bool allowFailure; bytes callData; }[] with one element, hand-encoded
        return abi.encode(_one(t, data));
    }
    struct Call3 { address target; bool allowFailure; bytes callData; }
    function _one(address t, bytes memory data) internal pure returns (Call3[] memory arr) { arr = new Call3[](1); arr[0] = Call3(t, false, data); }
    function _decodeAgg(bytes memory ret) internal pure returns (uint256) {
        (bool[] memory oks, bytes[] memory rets) = _decodeResults(ret);
        return oks[0] ? abi.decode(rets[0], (uint256)) : 0;
    }
    struct Result { bool success; bytes returnData; }
    function _decodeResults(bytes memory ret) internal pure returns (bool[] memory oks, bytes[] memory rets) {
        Result[] memory rs = abi.decode(ret, (Result[]));
        oks = new bool[](rs.length); rets = new bytes[](rs.length);
        for (uint256 i = 0; i < rs.length; i++) { oks[i] = rs[i].success; rets[i] = rs[i].returnData; }
    }

    // ---------- 5. inner-call revert: bundle does NOT revert; op is marked failed, nonce consumed, paymaster still charged ----------
    function test_innerRevert_isSoftFailure() public {
        delegatePayer();

        calls[0] = ISimple7702Account.Call(AUSD, 0, abi.encodeCall(IERC20.approve, (address(router), 1e6)));

        PackedUserOperation memory op = buildOp("", batchCalldata(calls));
        op = signOp(op, EP.getUserOpHash(op), pk);
        uint256 pmDep = EP.balanceOf(address(pm));
        OpResult memory r = submit(op);
        assertTrue(r.found, "event emitted"); assertFalse(r.success, "success == false");
        assertTrue(r.revertReasonEmitted, "UserOperationRevertReason emitted");
        console2.logBytes(r.revertReason); // ExecuteError(1, <inner revert>) from BaseAccount.executeBatch
        assertEq(bytes4(r.revertReason), bytes4(keccak256("ExecuteError(uint256,bytes)")));
        assertEq(router.settlements(), 0, "no settlement");
        assertEq(IERC20(AUSD).allowance(payer, address(router)), 0, "approve rolled back with the batch");
        assertEq(EP.getNonce(payer, 0), 1, "nonce consumed anyway");
        assertEq(pmDep - EP.balanceOf(address(pm)), r.actualGasCost, "paymaster still paid");
        assertEq(uint8(pm.lastMode()), uint8(IPaymaster.PostOpMode.opReverted));
    }
